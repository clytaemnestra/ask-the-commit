# INGEST-ONLY dependencies — used locally to turn audio into a committed index.
# Never installed on the deployed service.
#
#   pip install -r requirements-ingest.txt
#   python ingest.py          # audio -> transcript -> chunks -> data/index/
#   git add data/index && git commit
#
# NOTE: sentence-transformers pulls torch. On a CPU-only machine install the
# slim CPU wheel first to avoid ~2.5GB of unused CUDA libraries:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu

-r requirements.txt
